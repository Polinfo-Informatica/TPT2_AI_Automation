:name Toggle_F
:global int active
key.f()
gotoif(turn_off, active == 1)
active = 1
execute("Master")
goto(99)
turn_off:
active = 0
stop("Master")
stop("Auto_Mine")
stop("Auto_Shop")
stop("Auto_Base")
stop("Auto_Parts")
stop("Auto_Up1")
stop("Auto_Up2")
stop("Auto_Prod1")
stop("Auto_Prod2")

:name Master
cycle:
stop("Auto_Shop")
stop("Auto_Base")
stop("Auto_Parts")
stop("Auto_Up1")
stop("Auto_Up2")
stop("Auto_Prod1")
stop("Auto_Prod2")
show("factory", false)
show("mine", true)
execute("Auto_Mine")
wait(1.5)
stop("Auto_Mine")
show("mine", false)
show("factory", true)
execute("Auto_Shop")
execute("Auto_Base")
execute("Auto_Parts")
execute("Auto_Up1")
execute("Auto_Up2")
execute("Auto_Prod1")
execute("Auto_Prod2")
wait(45.0)
goto(cycle)

:name Auto_Mine
:local int y
newlayer()
y = 0
row_loop:
dig(0, y)
dig(1, y)
dig(2, y)
dig(3, y)
y = y + 1
gotoif(row_loop, y < 4)
